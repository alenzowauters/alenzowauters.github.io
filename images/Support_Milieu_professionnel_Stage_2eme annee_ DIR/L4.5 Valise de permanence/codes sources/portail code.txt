<script>
    // --- 1. GESTION DU POPUP ---
    function fermerModal() {
        document.getElementById('startModal').classList.remove('visible');
    }

    // --- RACCOURCI CLIC MOLETTE ---
    function handleMiddleClick(e) {
        if (e.button === 1) { // 1 = clic molette
            window.print();
        }
    }

    // --- 2. ANIMATION CANVAS ---
    const canvas = document.getElementById('bg-canvas');
    const ctx = canvas.getContext('2d');
    canvas.width = window.innerWidth; canvas.height = window.innerHeight;

    let lines = [];
    for(let i=0; i<40; i++) {
        lines.push({
            x: Math.random() * canvas.width, y: Math.random() * canvas.height,
            w: Math.random() * 100 + 50, s: Math.random() * 5 + 2, o: Math.random() * 0.5
        });
    }

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        const gradient = ctx.createLinearGradient(0, 0, 0, canvas.height);
        gradient.addColorStop(0, '#0f172a'); gradient.addColorStop(1, '#1e293b');
        ctx.fillStyle = gradient; ctx.fillRect(0,0, canvas.width, canvas.height);

        lines.forEach(l => {
            l.x += l.s; if(l.x > canvas.width) l.x = -l.w;
            ctx.fillStyle = `rgba(255, 255, 255, ${l.o * 0.1})`; ctx.fillRect(l.x, l.y, l.w, 2);
        });
        requestAnimationFrame(animate);
    }
    animate();

    // --- 3. LOGIQUE GRIST + SABLIER ---
    let records = [];
    grist.ready({ requiredAccess: 'full' });

    function startTimer() {
        const bar = document.getElementById('progress-bar');
        bar.style.transition = 'none'; bar.style.width = '100%';
        setTimeout(() => {
            bar.style.transition = 'width 5s linear'; bar.style.width = '0%';
        }, 50);
    }

    setInterval(() => {
        startTimer();
    }, 5000);

    grist.onRecords(r => {
        records = r;
        if(!document.getElementById('d1').value) {
            const now = new Date();
            document.getElementById('d1').valueAsDate = new Date(now.getFullYear(), now.getMonth(), 1);
            document.getElementById('d2').valueAsDate = now;
        }
        rafraichir();
        startTimer();
    });

    function forcerRafraichir() {
        rafraichir();
        startTimer();
    }

    function rafraichir() {
        const d1 = new Date(document.getElementById('d1').value);
        const d2 = new Date(document.getElementById('d2').value);
        d2.setHours(23, 59, 59);

        const data = records.filter(r => {
            const dateVal = r.Date || r.date; if (!dateVal) return false;
            const d = new Date(dateVal); return d >= d1 && d <= d2;
        }).sort((a,b) => new Date(b.Date||b.date) - new Date(a.Date||a.date));

        const container = document.getElementById('tableau-contenu');

        if(data.length === 0) {
            container.innerHTML = `<div style="text-align:center; padding:50px; color:white; width:100%;">🚫 Aucune donnée sur cette période.</div>`;
            return;
        }

        let html = '';
        data.forEach(r => {
            const flux = (r.Appel || r.appel || "").toString();
            const isEmis = flux.toLowerCase().includes('émis') || flux.toLowerCase().includes('emis');
            const cardClass = isEmis ? 'card-emis' : 'card-recu';
            const badgeClass = isEmis ? 'badge-emis' : 'badge-recu';
            const icon = isEmis ? 'fa-arrow-up' : 'fa-arrow-down';
            
            const info = r.Information || r.information || r.Info || "";
            const action = r.Action || r.action || "";
            const correspondant = r.Correspondant || r.correspondant || "Inconnu";
            const dateObj = new Date(r.Date||r.date);

            html += `
            <div class="reponse-card ${cardClass}">
                <div class="card-header">
                    <span class="card-time">${dateObj.toLocaleDateString('fr-FR')} - ${dateObj.toLocaleTimeString('fr-FR', {hour:'2-digit', minute:'2-digit'})}</span>
                    <span class="card-badge ${badgeClass}"><i class="fas ${icon}"></i> ${flux.toUpperCase()}</span>
                </div>
                <div class="card-corresp">${correspondant}</div>
                <div class="card-info">${info}</div>
                ${action ? `<div class="card-action"><strong><i class="fas fa-bolt"></i> Action :</strong> ${action}</div>` : ''}
                <div class="card-footer">Saisi par : ${r.Cadre||r.cadre||"?"}</div>
            </div>`;
        });

        container.innerHTML = html;
        document.getElementById('last-update').innerHTML = "🟢 MàJ : " + new Date().toLocaleTimeString();
    }

    // --- 4. EFFET TILT ---
    document.addEventListener('mousemove', (e) => {
        const card = document.querySelector('.js-tilt');
        const rect = card.getBoundingClientRect();
        const x = e.clientX - (rect.left + rect.width / 2);
        const y = e.clientY - (rect.top + rect.height / 2);
        card.style.transform = `perspective(1000px) rotateY(${x * 0.002}deg) rotateX(${-y * 0.002}deg)`;
    });

</script>
