    <script>
        // --- 1. CONFIGURATION GRIST ---
        grist.ready({
            columns: ['Cadre', 'Appel', 'Correspondant', 'Information', 'Action'],
            requiredAccess: 'none' // Plus besoin de clé API grâce aux permissions avancées
        });

        // --- 2. THEME & MODAL ---
        function switchTheme(type) {
            const body = document.body;
            if(type === 'recu') body.classList.add('theme-recu');
            else body.classList.remove('theme-recu');
        }

        function closeModal() {
            document.getElementById('modal-portfolio').classList.remove('show');
        }

        function resetForm() {
            document.getElementById('monFormulaire').reset();
            document.querySelector('input[value="Émis"]').checked = true;
            switchTheme('emis');
        }

        // --- 3. SOUMISSION ---
// Remplace la partie soumission par celle-ci pour tester
document.getElementById('monFormulaire').addEventListener('submit', async function(e) {
    e.preventDefault();
    const btn = document.querySelector('.btn-submit');
    btn.innerHTML = 'Envoi...';
    btn.disabled = true;

    try {
        // Ciblage direct par nom de table pour éviter les erreurs de changement de page
        const table = grist.getTable('MC_cadres');
        await table.create({ fields: {
            'Cadre': document.getElementById('cadre').value,
            'Appel': document.querySelector('input[name="appel"]:checked').value,
            'Correspondant': document.getElementById('correspondant').value,
            'Information': document.getElementById('information').value,
            'Action': document.getElementById('action').value
        }});

        document.getElementById('success-msg').classList.add('visible');
        setTimeout(() => {
            document.getElementById('success-msg').classList.remove('visible');
            resetForm();
            btn.innerHTML = 'TRANSMETTRE';
            btn.disabled = false;
        }, 2000);

    } catch (err) {
        console.error("Erreur détaillée Grist:", err);
        alert("Blocage Grist : " + err.message); // Ceci nous dira l'erreur exacte
        btn.innerHTML = 'ERREUR ❌';
        btn.disabled = false;
    }
});

        // --- 4. ANIMATIONS (PARTICULES & TILT) ---
        const canvas = document.getElementById('canvas-bg');
        const ctx = canvas.getContext('2d');
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;

        const particles = [];
        for(let i=0; i<60; i++) {
            particles.push({
                x: Math.random() * canvas.width,
                y: Math.random() * canvas.height,
                vx: (Math.random() - 0.5) * 0.5,
                vy: (Math.random() - 0.5) * 0.5,
                size: Math.random() * 2
            });
        }

        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
            particles.forEach((p, index) => {
                p.x += p.vx; p.y += p.vy;
                if(p.x < 0 || p.x > canvas.width) p.vx *= -1;
                if(p.y < 0 || p.y > canvas.height) p.vy *= -1;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fill();
            });
            requestAnimationFrame(animate);
        }
        animate();
        
        const card = document.querySelector('.js-tilt');
        document.addEventListener('mousemove', (e) => {
            const xAxis = (window.innerWidth / 2 - e.pageX) / 25;
            const yAxis = (window.innerHeight / 2 - e.pageY) / 25;
            card.style.transform = `rotateY(${xAxis}deg) rotateX(${yAxis}deg)`;
        });
    </script>
